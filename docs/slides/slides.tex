% ============================================================
\section{Introdução}
% ============================================================

\begin{frame}{Introdução e objetivos}
    \begin{itemize}
        \item Minas Gerais produz cerca de \alert{metade do café do país}.
        \item Café e soja são cotados em \textbf{dólar} e dependem do \textbf{clima}.
        \item Os dados são públicos, mas estão espalhados.
    \end{itemize}

    \vspace{0.2cm}
    \begin{block}{Objetivo}
        Reunir preço, câmbio e clima (2020--2026) em uma única base e analisar como se relacionam.
    \end{block}

    \begin{exampleblock}{Entregas}
        \textit{Crawler} diário, base integrada, gráficos, painel, modelos e repositório público.
    \end{exampleblock}
\end{frame}

\begin{frame}{Problema e perguntas de pesquisa}
    \begin{center}
        \small
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{@{}ll@{}}
            \toprule
            \textbf{P1} & Como os preços evoluíram? \\
            \textbf{P2} & Há sazonalidade de safra e entressafra? \\
            \textbf{P3} & Os preços acompanham o dólar? \\
            \textbf{P4} & Seca ou temperatura extrema antecede alta do café? \\
            \textbf{P5} & Qual foi o impacto da geada de 2021? \\
            \textbf{P6} & Quem é mais sensível ao clima e ao câmbio? \\
            \bottomrule
        \end{tabular}
    \end{center}

    \begin{exampleblock}{}
        A soja é a comparação: não depende de uma única região.
    \end{exampleblock}
\end{frame}

\section{Metodologia}
% ============================================================

\begin{frame}{Extração de dados}
    \centering
    \footnotesize
    TABELA I\\ \textsc{Fontes e técnicas de extração}\\[0.1cm]
    \begin{tabular}{@{}lll@{}}
        \toprule
        Fonte & Dado & Técnica \\
        \midrule
        Notícias Agrícolas & Café e soja & \textit{Web scraping} (\texttt{BeautifulSoup}) \\
        CEPEA/Esalq & Café e soja (prioritário) & Download manual \\
        Banco Central & Dólar e IPCA & API REST \\
        NASA POWER & Chuva e temperatura & API REST \\
        INMET & Estações (validação) & Download de CSV \\
        Yahoo Finance & Bolsas ICE e CBOT & \texttt{yfinance} \\
        \bottomrule
    \end{tabular}

    \vspace{0.2cm}
    \raggedright
    \begin{alertblock}{Ética}
        Respeito ao \texttt{robots.txt} e 3 s entre requisições. O CEPEA bloqueia robôs, por isso o download foi manual.
    \end{alertblock}
\end{frame}

\begin{frame}{Tipos dos dados e variáveis categóricas}
    \small
    \begin{center}
        {\footnotesize TABELA II\\ \textsc{Tipos das variáveis e tratamento}}\\[0.1cm]
        \begin{tabular}{@{}lll@{}}
            \toprule
            Tipo & Exemplo & O que foi feito \\
            \midrule
            Temporal & data & Índice de dias úteis \\
            Contínua & preço, chuva & Unidades padronizadas \\
            Discreta & dias secos & Contagem em 30 dias \\
            \midrule
            Nominal & região & Uma coluna por região \\
            Nominal & produto & Um modelo por produto \\
            Binária & safra/entressafra & Agrupamento \\
            Cíclica & mês & Seno e cosseno \\
            \bottomrule
        \end{tabular}
    \end{center}
\end{frame}

\begin{frame}{Tratamento e qualidade}
    \begin{itemize}
        \item Padronização com \texttt{pandas} e integração por dia útil
        \item \textit{Outliers} sinalizados por IQR e \alert{mantidos}
        \item Atributos defasados: chuva acumulada e dólar anterior
        \item Preços reais, deflacionados pelo IPCA
    \end{itemize}

    \begin{block}{Base integrada}
        \textbf{1.728 dias úteis}, sem ausentes em preço e dólar.
    \end{block}

    \begin{exampleblock}{Validação}
        \textit{Crawler} $\times$ CEPEA: diferença média quase nula.\\
        NASA $\times$ INMET: correlação de 0,86 a 0,95 na chuva mensal.
    \end{exampleblock}
\end{frame}

\section{Resultados}
% ============================================================

\begin{frame}{P1: como os preços evoluíram?}
    \centering
    \grafico{p1_evolucao_precos}

    \raggedright
    \begin{itemize}
        \item \textbf{Café}: de R\$ 534 para R\$ 1.709 (recorde de R\$ 2.769).
        \item \textbf{Soja}: pico de R\$ 207 em 2022 e R\$ 169 no fim.
    \end{itemize}
\end{frame}

\begin{frame}{P2: existe sazonalidade?}
    \begin{columns}[c]
        \begin{column}{0.60\textwidth}
            \grafico{p2_sazonalidade}
        \end{column}
        \begin{column}{0.37\textwidth}
            \begin{itemize}
                \item Café: menor em jun--jul ($\approx$94)
                \item Soja: menor em fev--jun (96--97)
                \item Kruskal-Wallis: $p<10^{-13}$, inflado pela dependência entre dias
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{P3: os preços acompanham o dólar?}
    \centering
    \grafico{p3_preco_dolar}

    \raggedright
    \begin{itemize}
        \item \textbf{Soja}: responde \alert{no mesmo dia} (correlação 0,34).
        \item \textbf{Café}: quase não responde (máximo de 0,06).
    \end{itemize}
\end{frame}

\begin{frame}{P4: o clima antecede altas do café?}
    \begin{columns}[c]
        \begin{column}{0.60\textwidth}
            \grafico{p4_clima_regioes}
        \end{column}
        \begin{column}{0.37\textwidth}
            \footnotesize
            Clima no Cerrado Mineiro $\times$ variação do preço em 3 meses (Spearman):

            \vspace{0.15cm}
            \begin{tabular}{@{}lr@{}}
                \toprule
                Máximas (30 d) & $+0{,}34$ \\
                Chuva (90 d) & $-0{,}26$ \\
                \bottomrule
            \end{tabular}

            \vspace{0.2cm}
            \alert{Calor e seca tendem a anteceder altas} (associação moderada).
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{P5: qual foi o impacto da geada de 2021?}
    \begin{columns}[c]
        \begin{column}{0.58\textwidth}
            \grafico{p5_geada_2021}
        \end{column}
        \begin{column}{0.39\textwidth}
            \centering\footnotesize
            Preço médio após a geada\\(base: 30 dias antes)

            \vspace{0.1cm}
            \setlength{\tabcolsep}{3pt}
            \begin{tabular}{@{}lrr@{}}
                \toprule
                & Café & Soja \\
                \midrule
                30 dias & +20,9\% & +5,9\% \\
                90 dias & +28,4\% & +6,3\% \\
                180 dias & \textbf{+46,9\%} & +6,1\% \\
                \bottomrule
            \end{tabular}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{exampleblock}{}
        Compatível com efeito climático, mas não prova causalidade.
    \end{exampleblock}
\end{frame}

\begin{frame}{P6: clima ou câmbio?}
    \begin{columns}[c]
        \begin{column}{0.56\textwidth}
            \grafico{p6_importancia_grupos}
        \end{column}
        \begin{column}{0.42\textwidth}
            \centering\scriptsize
            Peso do clima no modelo

            \vspace{0.1cm}
            \setlength{\tabcolsep}{3pt}
            \begin{tabular}{@{}lrr@{}}
                \toprule
                & Linear & RF \\
                \midrule
                Café, nível & 0,65 & 0,49 \\
                Soja, nível & 0,70 & 0,76 \\
                Café, var. 21 d & 0,77 & 0,83 \\
                Soja, var. 21 d & 0,72 & 0,80 \\
                \bottomrule
            \end{tabular}
        \end{column}
    \end{columns}

    \vspace{0.2cm}
    \begin{itemize}
        \item \textbf{Câmbio}: soja mais sensível no mesmo dia (0,34 $\times$ 0,05).
        \item \textbf{Clima}: a geada aponta para o café, mas os \alert{modelos são inconclusivos}.
    \end{itemize}
\end{frame}

\begin{frame}{Desempenho dos modelos}
    \centering
    \footnotesize
    TABELA III\\ \textsc{Erro fora da amostra (média de 5 folds)}\\[0.1cm]
    \begin{tabular}{@{}lrrrr@{}}
        \toprule
        & \multicolumn{3}{c}{MAE} & $R^2$ \\
        \cmidrule(lr){2-4}
        Produto e alvo & \textit{Baseline} & Linear & RF & RF \\
        \midrule
        Café, nível (R\$) & \textbf{532} & 595 & 624 & $-9{,}8$ \\
        Café, var. 21 d (p.p.) & \textbf{8,1} & 9,8 & 8,8 & $-0{,}40$ \\
        Soja, nível (R\$) & 23,1 & 28,1 & \textbf{23,0} & $-11{,}1$ \\
        Soja, var. 21 d (p.p.) & \textbf{3,9} & 4,8 & 4,8 & $-0{,}99$ \\
        \bottomrule
    \end{tabular}

    \vspace{0.2cm}
    \raggedright
    \begin{itemize}
        \item Quase sempre, o \textit{baseline} erra menos; $R^2$ de teste negativo.
        \item Cada teste cai em um \alert{regime de preços novo}.
    \end{itemize}
\end{frame}

\begin{frame}{Conclusão}
    \begin{block}{O que os dados mostram}
        O café mais que triplicou e subiu 46,9\% em 180 dias após a geada. A soja acompanha o dólar no mesmo dia.
    \end{block}

    \begin{columns}[T,onlytextwidth]
        \begin{column}{0.48\textwidth}
            \begin{alertblock}{Limitações}
                \footnotesize
                \begin{itemize}
                    \item Correlação não é causalidade
                    \item Um ponto de clima por região
                    \item Sem estoques e oferta mundial
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.48\textwidth}
            \begin{exampleblock}{Próximos passos}
                \footnotesize

                    \item Estoques e produção mundial
                    \item Clima das regiões da soja
                    \item Modelos de séries temporais
                \end{itemize}
            \end{exampleblock}
